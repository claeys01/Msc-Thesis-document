% figures/Methodology/tikz_figures/retrain_flowchart.tex
%
% Adaptive retraining flowchart with parallel CFD branch.
% Once the OOD flag triggers retraining, two branches execute concurrently:
%   - LEFT  : WaterLily continues advancing the simulation
%   - RIGHT : the ML pipeline (AE, NODE, KNN) is refit
% Both branches rejoin at the resume node.
%


\begin{tikzpicture}[
    % ===================================================================
    % STYLES — edit here to restyle the whole flowchart
    % ===================================================================
    node distance      = 6mm and 11mm,
    font               = \scriptsize,
    block/.style = {
    draw=black, thick, line width=0.4pt, 
    rounded corners=0.0pt,
    minimum height=8mm,
    align=center,
    inner sep=4pt, outer sep=0pt,
    font=\sffamily\itshape\scriptsize,
    blur shadow={shadow blur steps=4, shadow blur extra rounding=0.5pt,
                 shadow xshift=1pt, shadow yshift=-1pt, shadow opacity=25}
    },
    flag/.style        = {block,  minimum width=46mm, fill=red!14},
    cfdproc/.style     = {block,  minimum width=42mm, fill=warmupgreen},
    mlproc/.style      = {block,  minimum width=42mm, fill=nodegrey},
    retrainproc/.style = {block,  minimum width=42mm, fill=yellow!40!white},
                          %
    startstop/.style   = {ellipse, draw, thick, align=center, fill=gray!12,
                          minimum height=7mm,  minimum width=34mm},
                          %
    parallelbox/.style = {draw=black!55, dashed, rounded corners=4pt,
                          inner sep=4mm, fill=phaseretrain},
    % --- arrows / labels ---
    flow/.style        = {-{Latex[length=2mm,width=1.6mm]}, thick},
    line/.style        = {thick},  % no arrowhead, used for fork/join legs
    panellab/.style    = {font=\scriptsize\itshape, anchor=south east, inner sep=4pt},
]

% \newlength{\chainheight}
% \pgfmathsetlength{\chainheight}{4*8mm + 3*6mm}  % 4 nodes, 3 gaps

% =======================================================================
% TOP SPINE  — trigger and data-extension cutoff
% =======================================================================
\node[flag]                       (trigger) {%
    Retraining flag raised\\
    $n_\text{flag} \geq n_\text{max, flags}$};

\node[cfdproc, below=of trigger]  (cutoff)  {%
    Extend CFD trajectory\\
    \texttt{run\_cutoff}, duration $t^*_\text{update}$};

% =======================================================================
% RIGHT COLUMN  — pipeline retraining chain (declare FIRST)
% =======================================================================
\node[retrainproc, below right=14mm and 12mm of cutoff] (refitAE) {%
    Fine-tune autoencoder\\
    $\varphi_\theta^{(k)},\psi_\theta^{(k)} \to \varphi_\theta^{(k+1)},\psi_\theta^{(k+1)}$};

\node[mlproc, below=of refitAE]        (encode)    {%
    Re-encode training set\\
    $\mathbf{z}_n = \varphi_\theta^{(k+1)}(\mathbf{x}_{\text{in},n})$};

\node[retrainproc, below=of encode]    (refitNODE) {%
    Fine-tune neural ODE\\
    $f_\phi^{(k)} \to f_\phi^{(k+1)}$};

\node[retrainproc, below=of refitNODE] (refitKNN)  {%
    Refit KNN reference set\\
    $\mathcal{K}^{(k+1)},\,\tau^{(k+1)}$};

% =======================================================================
% LEFT COLUMN  — WaterLily continues in parallel
% Sized automatically to match the right column's vertical extent.
% =======================================================================
\node[cfdproc,
      fit={(refitAE.north -| refitAE.west) (refitKNN.south -| refitKNN.east)},
      text width=34mm,
      align=center,
      anchor=north east,
      inner sep=0pt]
  (parallel) at ([xshift=-14mm]refitAE.north west)
  {WaterLily continues\\advancing $\mathcal{S}$\\[3mm]
   \textit{conventional}\\\textit{CFD time-stepping,}\\
   \textit{no snapshot}\\\textit{collection}};

% =======================================================================
% RESUME NODE  — centred below both columns
% =======================================================================
\node[startstop, anchor=north]
  (resume) at ($(parallel.south east)!0.5!(refitKNN.south west) + (0,-14mm)$)
  {Resume hybrid loop\\reset $n_\text{flag},\,n_\text{step}$};

% =======================================================================
% ARROWS
% =======================================================================
% Top spine
\draw[flow] (trigger) -- (cutoff);

% Fork from cutoff into the two parallel branches
\coordinate (fork) at ([yshift=-6mm]cutoff.south);
\draw[line] (cutoff.south) -- (fork);
\draw[flow] (fork) -| (parallel.north);
\draw[flow] (fork) -| (refitAE.north);

% Right-column chain
\draw[flow] (refitAE)   -- (encode);
\draw[flow] (encode)    -- (refitNODE);
\draw[flow] (refitNODE) -- (refitKNN);

% Join into resume
\coordinate (joinpt) at ([yshift=6mm]resume.north);
\draw[line] (parallel.south) -- ++(0,-7mm) -| (joinpt);
\draw[line] (refitKNN.south) -- ++(0,-7mm) -| (joinpt);
\draw[flow] (joinpt) -- (resume.north);

% =======================================================================
% DASHED BACKGROUND PANEL highlighting the concurrent section
% =======================================================================
\begin{scope}[on background layer]
  \node[parallelbox, fit=(parallel)(refitAE)(refitKNN)] (parbox) {};
\end{scope}
\node[panellab] at (parbox.north east) {concurrent};

\end{tikzpicture}
